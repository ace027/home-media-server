  ```yaml
  services:
    base:
      restart: unless-stopped
      security_opt:
        - no-new-privileges:true
      environment:
        TZ: ${TZ}
        PUID: ${PUID}
        PGID: ${PGID}
        UMASK: ${UMASK}

        driver: json-file
        options:
          max-size: "10m"
          max-file: "3"
  ```

    - download: Phase 2 (sabnzbd)
    - transcode: Phase 4 (fileflows)
    - ops: Phase 5 (homepage, uptime-kuma, notifiarr, diun, backrest, dozzle)
  - A comment line: `# New services: extends: {file: _common.yaml, service: base} and pin image tags (never :latest).`
  - Then:
    ```yaml
    services: {}
    networks:
      proxy:
        name: proxy
        external: true
    ```
